
    // =====================================================================================
    // [B6] Diagnostics (staged on top of B5). Nothing here changes what a mod, a hook or
    // the engine does: counters, one RAII wrapper that tracks the lock depth, and log
    // lines. Everything between the CORE markers uses only the standard library plus the
    // four glue functions declared at its top, so the host tests compile exactly this text
    // (outputs/.../b6-diagnostics/tests/extract_core.py slices it out of the staged file).
    //
    //   [LOCKLEAK]  a Lua-callable function returned or raised while a lock scope opened
    //               inside it was still alive (longjmp skipped the destructor)
    //   [LOCKWAIT]  a lock acquisition waited more than 250 ms, or is still waiting
    //   [PERF]      once per second from the game-thread pump
    //   [l22] widget/object construct summary   (rate limiter for the per-widget line)
    // =====================================================================================
    // [B6-CORE-BEGIN]
    namespace
    {
    namespace b6diag
    {
        using i64 = long long;

        // ---- glue, defined after this region (the tests define their own) -----------
        auto emit_line(const char* line) -> void;   // one log line, no trailing newline
        auto on_game_thread() -> bool;              // true on the published game thread

        constexpr i64 kLockWaitLogMs = 250;         // log an acquisition that waited longer
        constexpr int kWaitSlots = 8;
        constexpr int kMaxMods = 16;                // slot kMaxMods is the "?" slot
        constexpr int kWidgetLinesPerSecond = 5;

        inline auto now_ms() noexcept -> i64
        {
            return std::chrono::duration_cast<std::chrono::milliseconds>(std::chrono::steady_clock::now().time_since_epoch()).count();
        }
        inline auto now_us() noexcept -> i64
        {
            return std::chrono::duration_cast<std::chrono::microseconds>(std::chrono::steady_clock::now().time_since_epoch()).count();
        }

        // Copies up to n-1 chars, control characters (newlines of multi-line Lua
        // errors) become spaces, always NUL-terminated.
        inline auto copy_one_line(char* out, size_t n, const char* text) noexcept -> void
        {
            if (out == nullptr || n == 0) { return; }
            size_t i = 0;
            if (text != nullptr)
            {
                for (; i + 1 < n && text[i] != '\0'; ++i)
                {
                    const unsigned char c = static_cast<unsigned char>(text[i]);
                    out[i] = (c < 0x20 || c == 0x7f) ? ' ' : static_cast<char>(c);
                }
            }
            out[i] = '\0';
        }

        // ---- rate limiter: at most `limit` verbatim lines per window --------------
        // Events past the limit are counted. When a window ends, the verdict of the
        // next admit()/flush() carries the finished window's totals, so the caller
        // can print ONE summary line. Thread safe (own mutex, never held while logging).
        struct RateLimiter
        {
            struct Verdict
            {
                bool show{false};
                int prev_total{0};       // events in the window that just ended
                int prev_suppressed{0};  // of those, not shown (0 = no summary needed)
            };
            std::mutex mutex{};
            bool started{false};
            i64 window_start_ms{0};
            int total{0};
            int shown{0};

            auto roll_locked(i64 now, i64 window_ms) -> Verdict
            {
                Verdict v{};
                if (!started || now - window_start_ms >= window_ms)
                {
                    if (started && total > shown)
                    {
                        v.prev_total = total;
                        v.prev_suppressed = total - shown;
                    }
                    started = true;
                    window_start_ms = now;
                    total = 0;
                    shown = 0;
                }
                return v;
            }
            auto admit(i64 now, int limit, i64 window_ms) -> Verdict
            {
                std::lock_guard<std::mutex> guard{mutex};
                Verdict v = roll_locked(now, window_ms);
                ++total;
                if (shown < limit)
                {
                    ++shown;
                    v.show = true;
                }
                return v;
            }
            // Ends an expired window without counting an event (called from the pump).
            auto flush(i64 now, i64 window_ms) -> Verdict
            {
                std::lock_guard<std::mutex> guard{mutex};
                return roll_locked(now, window_ms);
            }
        };

        // ---- per-mod table (key = the mod's main lua_State) ------------------------
        struct ModSlot
        {
            std::atomic<lua_State*> key{nullptr};
            char name[40]{};
            std::atomic<i64> game_us{0};          // game-thread Lua time (outermost call_function)
            std::atomic<i64> game_calls{0};
            i64 prev_game_us{0};                  // pump bookkeeping (pump thread only)
            i64 prev_game_calls{0};
            std::atomic<i64> leak_total{0};
            std::atomic<i64> leak_unlogged{0};    // leaks since the last [LOCKLEAK] line for this mod
            std::atomic<i64> leak_last_log_ms{0};
        };
        inline ModSlot g_mods[kMaxMods + 1];
        inline std::atomic<int> g_mod_count{0};
        inline std::atomic<i64> g_leak_total{0};

        inline auto register_mod(lua_State* key, const char* ascii_name) -> void
        {
            if (key == nullptr) { return; }
            const int have = std::min(g_mod_count.load(std::memory_order_acquire), kMaxMods);
            for (int i = 0; i < have; ++i)
            {
                if (g_mods[i].key.load(std::memory_order_acquire) == key) { return; }
            }
            const int index = g_mod_count.fetch_add(1, std::memory_order_acq_rel);
            if (index >= kMaxMods) { return; }
            copy_one_line(g_mods[index].name, sizeof(g_mods[index].name), ascii_name != nullptr ? ascii_name : "?");
            g_mods[index].key.store(key, std::memory_order_release);
        }
        inline auto find_mod(lua_State* key) -> ModSlot&
        {
            const int have = std::min(g_mod_count.load(std::memory_order_acquire), kMaxMods);
            for (int i = 0; i < have; ++i)
            {
                if (g_mods[i].key.load(std::memory_order_acquire) == key && key != nullptr) { return g_mods[i]; }
            }
            if (g_mods[kMaxMods].name[0] == '\0') { copy_one_line(g_mods[kMaxMods].name, sizeof(g_mods[kMaxMods].name), "?"); }
            return g_mods[kMaxMods];
        }

        // ---- [LOCKLEAK] -------------------------------------------------------------
        // Decides whether this mod's leak is logged now: the first one always, later
        // ones at most every `gap_ms`, with the number skipped since the last line.
        struct LeakVerdict
        {
            bool log{false};
            i64 repeats{0};  // leaks not logged since the previous line, including this one
        };
        inline auto leak_decide(ModSlot& slot, i64 now, i64 gap_ms) -> LeakVerdict
        {
            slot.leak_total.fetch_add(1, std::memory_order_relaxed);
            const i64 pending = slot.leak_unlogged.fetch_add(1, std::memory_order_relaxed) + 1;
            i64 last = slot.leak_last_log_ms.load(std::memory_order_relaxed);
            if (last == 0 || now - last >= gap_ms)
            {
                if (slot.leak_last_log_ms.compare_exchange_strong(last, now, std::memory_order_relaxed))
                {
                    slot.leak_unlogged.store(0, std::memory_order_relaxed);
                    return LeakVerdict{true, pending};
                }
            }
            return LeakVerdict{false, pending};
        }
        // Hook installed into LuaMadeSimple (see DiagHooks). Never throws.
        inline auto on_leak(lua_State* main_state, const char* kind, const char* fn, const char* text, int extra, int depth) noexcept -> void
        {
            try
            {
                ModSlot& slot = find_mod(main_state);
                g_leak_total.fetch_add(1, std::memory_order_relaxed);
                const LeakVerdict verdict = leak_decide(slot, now_ms(), 30000);
                if (!verdict.log) { return; }
                char one_line[160];
                copy_one_line(one_line, sizeof(one_line), text);
                char line[480];
                std::snprintf(line, sizeof(line), "[LOCKLEAK] mod=%s fn=%s kind=%s extra=%d depth=%d repeats=%lld err=%s",
                              slot.name, (fn != nullptr && fn[0] != '\0') ? fn : "?", kind != nullptr ? kind : "?", extra, depth, verdict.repeats, one_line);
                emit_line(line);
            }
            catch (...)
            {
            }
        }
        // The pump runs on the game thread outside any dispatch. A depth above zero
        // there that does not change between two pumps is a skipped lock scope.
        inline auto pump_depth_stuck(int depth_now, bool dispatching, int& last_depth) noexcept -> bool
        {
            const bool stuck = depth_now > 0 && !dispatching && depth_now == last_depth;
            last_depth = (depth_now > 0 && !dispatching) ? depth_now : 0;
            return stuck;
        }

        // ---- game-thread Lua time per mod (hooks for LuaMadeSimple::call_function) ----
        inline auto hook_call_begin() noexcept -> long long { return on_game_thread() ? now_us() : 0; }
        inline auto hook_call_end(lua_State* main_state, long long begin_us) noexcept -> void
        {
            ModSlot& slot = find_mod(main_state);
            slot.game_us.fetch_add(now_us() - begin_us, std::memory_order_relaxed);
            slot.game_calls.fetch_add(1, std::memory_order_relaxed);
        }
        inline auto install_hooks_once() -> void
        {
            static const bool installed = []() {
                LuaMadeSimple::DiagHooks hooks{};
                hooks.on_leak = &on_leak;
                hooks.call_begin = &hook_call_begin;
                hooks.call_end = &hook_call_end;
                LuaMadeSimple::diag_install_hooks(hooks);
                return true;
            }();
            (void)installed;
        }

        // ---- lock tracking ---------------------------------------------------------
        struct WaitSlot
        {
            std::atomic<i64> since_ms{0};   // 0 = free, -1 = being claimed
            std::atomic<int> line{0};
            std::atomic<int> game{0};
            std::atomic<i64> last_log_ms{0};
        };
        struct Holder
        {
            std::atomic<i64> since_ms{0};   // 0 = nobody (outermost scope of one thread)
            std::atomic<int> line{0};
            std::atomic<int> game{0};
        };
        inline WaitSlot g_wait[kWaitSlots];
        inline Holder g_holder;
        inline std::atomic<i64> g_slow_acquires{0};
        inline std::atomic<i64> g_slow_max_ms{0};      // since the last [PERF] line
        inline std::atomic<i64> g_lockwait_events{0};  // acquisitions over kLockWaitLogMs
        inline RateLimiter g_lockwait_limiter{};

        inline auto wait_begin(i64 t0, int line, int game) noexcept -> int
        {
            for (int i = 0; i < kWaitSlots; ++i)
            {
                i64 expected = 0;
                if (g_wait[i].since_ms.compare_exchange_strong(expected, -1, std::memory_order_acq_rel))
                {
                    g_wait[i].line.store(line, std::memory_order_relaxed);
                    g_wait[i].game.store(game, std::memory_order_relaxed);
                    g_wait[i].last_log_ms.store(0, std::memory_order_relaxed);
                    g_wait[i].since_ms.store(t0, std::memory_order_release);
                    return i;
                }
            }
            return -1;
        }
        inline auto thread_word(int game) noexcept -> const char* { return game != 0 ? "game" : "other"; }
        inline auto log_lockwait(i64 waited_ms, int line, int game) -> void
        {
            const auto verdict = g_lockwait_limiter.admit(now_ms(), 5, 10000);
            char buffer[200];
            if (verdict.prev_suppressed > 0)
            {
                std::snprintf(buffer, sizeof(buffer), "[LOCKWAIT] summary: %d acquisitions over %lld ms in the last window, %d lines suppressed",
                              verdict.prev_total, static_cast<i64>(kLockWaitLogMs), verdict.prev_suppressed);
                emit_line(buffer);
            }
            if (verdict.show)
            {
                std::snprintf(buffer, sizeof(buffer), "[LOCKWAIT] site_line=%d thread=%s waited_ms=%lld", line, thread_word(game), waited_ms);
                emit_line(buffer);
            }
        }
        // Slow path of the acquisition: only reached when try_lock failed, so the
        // uncontended path never reads a clock.
        inline auto lock_slow(std::recursive_mutex& mutex, int line) -> void
        {
            const i64 t0 = now_ms();
            const int game = on_game_thread() ? 1 : 0;
            const int slot = wait_begin(t0, line, game);
            mutex.lock();
            if (slot >= 0) { g_wait[slot].since_ms.store(0, std::memory_order_release); }
            const i64 waited = now_ms() - t0;
            g_slow_acquires.fetch_add(1, std::memory_order_relaxed);
            i64 seen = g_slow_max_ms.load(std::memory_order_relaxed);
            while (waited > seen && !g_slow_max_ms.compare_exchange_weak(seen, waited, std::memory_order_relaxed)) {}
            if (waited > kLockWaitLogMs)
            {
                g_lockwait_events.fetch_add(1, std::memory_order_relaxed);
                log_lockwait(waited, line, game);
            }
        }
        inline auto holder_set(int line) -> void
        {
            g_holder.line.store(line, std::memory_order_relaxed);
            g_holder.game.store(on_game_thread() ? 1 : 0, std::memory_order_relaxed);
            g_holder.since_ms.store(now_ms(), std::memory_order_release);
        }
        inline auto holder_clear() noexcept -> void { g_holder.since_ms.store(0, std::memory_order_release); }

        // Replaces std::lock_guard<std::recursive_mutex> on LuaMod::m_thread_actions_mutex.
        // Same locking behaviour. Adds: the per-thread scope depth (a skipped destructor
        // leaves it too high), the wait bookkeeping above, and the "who holds it" record.
        // `line` is __LINE__ of the declaration (site id, resolved with the staged file).
        class ActionsLock
        {
            std::recursive_mutex& m_mutex;
        public:
            ActionsLock(std::recursive_mutex& mutex, int line) : m_mutex(mutex)
            {
                if (!m_mutex.try_lock()) { lock_slow(m_mutex, line); }
                if (LuaMadeSimple::diag_scope_enter() == 0) { holder_set(line); }
            }
            ~ActionsLock()
            {
                LuaMadeSimple::diag_scope_exit();
                if (LuaMadeSimple::diag_scope_depth() == 0) { holder_clear(); }
                m_mutex.unlock();
            }
            ActionsLock(const ActionsLock&) = delete;
            auto operator=(const ActionsLock&) -> ActionsLock& = delete;
        };

        // Lines for waiters stuck longer than kLockWaitLogMs, at most one per slot every
        // 5 s. Called from the pump, so a lock that never comes back is still reported.
        inline auto scan_stuck_waiters(i64 now, char (*lines)[260], int max_lines) -> int
        {
            int produced = 0;
            const i64 holder_since = g_holder.since_ms.load(std::memory_order_acquire);
            for (int i = 0; i < kWaitSlots && produced < max_lines; ++i)
            {
                const i64 since = g_wait[i].since_ms.load(std::memory_order_acquire);
                if (since <= 0 || now - since <= kLockWaitLogMs) { continue; }
                const i64 last = g_wait[i].last_log_ms.load(std::memory_order_relaxed);
                if (last != 0 && now - last < 5000) { continue; }
                g_wait[i].last_log_ms.store(now, std::memory_order_relaxed);
                if (holder_since > 0)
                {
                    std::snprintf(lines[produced], sizeof(lines[produced]),
                                  "[LOCKWAIT] still waiting site_line=%d thread=%s waited_ms=%lld holder_line=%d holder_thread=%s holder_held_ms=%lld",
                                  g_wait[i].line.load(std::memory_order_relaxed), thread_word(g_wait[i].game.load(std::memory_order_relaxed)), now - since,
                                  g_holder.line.load(std::memory_order_relaxed), thread_word(g_holder.game.load(std::memory_order_relaxed)), now - holder_since);
                }
                else
                {
                    std::snprintf(lines[produced], sizeof(lines[produced]),
                                  "[LOCKWAIT] still waiting site_line=%d thread=%s waited_ms=%lld holder=none-recorded",
                                  g_wait[i].line.load(std::memory_order_relaxed), thread_word(g_wait[i].game.load(std::memory_order_relaxed)), now - since);
                }
                ++produced;
            }
            return produced;
        }

        // ---- FindFirstOf / FindAllOf, widget constructions ---------------------------
        inline std::atomic<i64> g_find_first_calls{0};
        inline std::atomic<i64> g_find_first_us{0};
        inline std::atomic<i64> g_find_all_calls{0};
        inline std::atomic<i64> g_find_all_us{0};
        inline std::atomic<i64> g_widgets{0};
        inline RateLimiter g_widget_limiter{};

        // Counts the call on entry and adds its duration on exit. A Lua raise skips the
        // destructor: the call is counted, its time is not. Nothing else is held.
        class FindTimer
        {
            std::atomic<i64>& m_us;
            i64 m_t0;
        public:
            FindTimer(std::atomic<i64>& calls, std::atomic<i64>& us) noexcept : m_us(us), m_t0(now_us()) { calls.fetch_add(1, std::memory_order_relaxed); }
            ~FindTimer() { m_us.fetch_add(now_us() - m_t0, std::memory_order_relaxed); }
            FindTimer(const FindTimer&) = delete;
            auto operator=(const FindTimer&) -> FindTimer& = delete;
        };

        inline auto widget_gate(i64 now) -> RateLimiter::Verdict
        {
            g_widgets.fetch_add(1, std::memory_order_relaxed);
            return g_widget_limiter.admit(now, kWidgetLinesPerSecond, 1000);
        }
        inline auto emit_widget_summary(const RateLimiter::Verdict& verdict) -> void
        {
            if (verdict.prev_suppressed <= 0) { return; }
            char line[200];
            std::snprintf(line, sizeof(line), "[l22] widget/object construct summary: %d constructed in the last window, %d log lines suppressed",
                          verdict.prev_total, verdict.prev_suppressed);
            emit_line(line);
        }

        // ---- [PERF] ----------------------------------------------------------------
        struct QueueDepths
        {
            bool ok{false};       // false = the lock was busy, sizes not read
            size_t game{0}, game_pending{0}, delayed{0}, delayed_pending{0}, tick{0}, tick_pending{0};
        };
        struct PerfInput
        {
            struct ModRow { const char* name; i64 us; i64 calls; };
            i64 dt_ms{0};
            ModRow rows[kMaxMods + 1]{};
            int row_count{0};
            i64 ff_calls{0}, ff_us{0}, fa_calls{0}, fa_us{0};
            i64 widgets{0};
            QueueDepths queues{};
            i64 slow_acquires{0}, slow_max_ms{0}, lockwait_events{0};
            i64 leaks{0};
            int pump_depth{0};
        };
        // Bounded text builder: every append is a literal-format snprintf, the
        // buffer is always NUL-terminated and never overrun.
        struct TextBuf
        {
            char* out;
            size_t n;
            size_t used{0};
            auto room() const noexcept -> size_t { return used < n ? n - used : 0; }
            auto tail() noexcept -> char* { return out + (used < n ? used : n - 1); }
            auto adv(int wrote) noexcept -> void
            {
                if (wrote > 0) { used = std::min(n - 1, used + static_cast<size_t>(wrote)); }
            }
        };
        // "x.y" milliseconds from microseconds.
        inline auto put_ms(TextBuf& b, i64 us) noexcept -> void
        {
            b.adv(std::snprintf(b.tail(), b.room(), "%lld.%lld", us / 1000, (us % 1000) / 100));
        }
        // Always NUL-terminated, never writes past n. Returns the length.
        inline auto format_perf(char* out, size_t n, const PerfInput& in) noexcept -> size_t
        {
            if (out == nullptr || n == 0) { return 0; }
            out[0] = '\0';
            TextBuf b{out, n};
            b.adv(std::snprintf(b.tail(), b.room(), "[PERF] dt=%lldms lua_ms={", in.dt_ms));
            bool any = false;
            for (int i = 0; i < in.row_count; ++i)
            {
                if (in.rows[i].calls <= 0) { continue; }
                b.adv(std::snprintf(b.tail(), b.room(), "%s%s:", any ? "," : "", in.rows[i].name != nullptr ? in.rows[i].name : "?"));
                put_ms(b, in.rows[i].us);
                b.adv(std::snprintf(b.tail(), b.room(), "/%lld", in.rows[i].calls));
                any = true;
            }
            b.adv(std::snprintf(b.tail(), b.room(), "%s} find_first=%lld/", any ? "" : "-", in.ff_calls));
            put_ms(b, in.ff_us);
            b.adv(std::snprintf(b.tail(), b.room(), "ms find_all=%lld/", in.fa_calls));
            put_ms(b, in.fa_us);
            b.adv(std::snprintf(b.tail(), b.room(), "ms widgets=%lld", in.widgets));
            if (in.queues.ok)
            {
                b.adv(std::snprintf(b.tail(), b.room(), " q=g%zu,p%zu,d%zu,pd%zu,e%zu,pe%zu", in.queues.game, in.queues.game_pending, in.queues.delayed,
                                    in.queues.delayed_pending, in.queues.tick, in.queues.tick_pending));
            }
            else
            {
                b.adv(std::snprintf(b.tail(), b.room(), " q=busy"));
            }
            b.adv(std::snprintf(b.tail(), b.room(), " lockwait=%lld/max%lldms/over%lld leaks=%lld pump_depth=%d", in.slow_acquires, in.slow_max_ms,
                                in.lockwait_events, in.leaks, in.pump_depth));
            return b.used;
        }
        // Fills `in` from the live counters and moves the "since last line" baselines.
        // Pump thread only (the baselines are plain).
        inline auto collect_perf(PerfInput& in, i64 now, i64& last_ms, const QueueDepths& queues, int pump_depth) -> void
        {
            static i64 prev_ff_calls = 0, prev_ff_us = 0, prev_fa_calls = 0, prev_fa_us = 0, prev_widgets = 0;
            in.dt_ms = last_ms == 0 ? 0 : now - last_ms;
            last_ms = now;
            const int have = std::min(g_mod_count.load(std::memory_order_acquire), kMaxMods);
            in.row_count = 0;
            for (int i = 0; i <= kMaxMods; ++i)
            {
                if (i >= have && i != kMaxMods) { continue; }
                ModSlot& slot = g_mods[i];
                const i64 us = slot.game_us.load(std::memory_order_relaxed);
                const i64 calls = slot.game_calls.load(std::memory_order_relaxed);
                in.rows[in.row_count++] = PerfInput::ModRow{slot.name[0] != '\0' ? slot.name : "?", us - slot.prev_game_us, calls - slot.prev_game_calls};
                slot.prev_game_us = us;
                slot.prev_game_calls = calls;
            }
            const i64 ff_calls = g_find_first_calls.load(std::memory_order_relaxed), ff_us = g_find_first_us.load(std::memory_order_relaxed);
            const i64 fa_calls = g_find_all_calls.load(std::memory_order_relaxed), fa_us = g_find_all_us.load(std::memory_order_relaxed);
            const i64 widgets = g_widgets.load(std::memory_order_relaxed);
            in.ff_calls = ff_calls - prev_ff_calls; in.ff_us = ff_us - prev_ff_us;
            in.fa_calls = fa_calls - prev_fa_calls; in.fa_us = fa_us - prev_fa_us;
            in.widgets = widgets - prev_widgets;
            prev_ff_calls = ff_calls; prev_ff_us = ff_us; prev_fa_calls = fa_calls; prev_fa_us = fa_us; prev_widgets = widgets;
            in.queues = queues;
            in.slow_acquires = g_slow_acquires.exchange(0, std::memory_order_relaxed);
            in.slow_max_ms = g_slow_max_ms.exchange(0, std::memory_order_relaxed);
            in.lockwait_events = g_lockwait_events.load(std::memory_order_relaxed);
            in.leaks = g_leak_total.load(std::memory_order_relaxed);
            in.pump_depth = pump_depth;
        }
        // True once per `min_gap_ms` (both pumps may call; only one line per second).
        inline auto perf_gate(std::atomic<i64>& last, i64 now, i64 min_gap_ms) noexcept -> bool
        {
            i64 seen = last.load(std::memory_order_relaxed);
            return now - seen >= min_gap_ms && last.compare_exchange_strong(seen, now, std::memory_order_relaxed);
        }
    }
    }
    // [B6-CORE-END]
